<script>

    // 1. Core Mathematical Logistic Engine

    function f(x, r) { return r * x * (1.0 - x); }


    // 2. Pre-calculate static background Bifurcation data points for fast execution

    const bifR = [];

    const bifX = [];

    const nR = 800; 

    const nIters = 400;

    const nLast = 100;


    for (let i = 0; i < nR; i++) {

        let r = 2.4 + (i / nR) * 1.6;

        let x = 0.5;

        for (let j = 0; j < nIters; j++) {

            x = f(x, r);

            if (j >= (nIters - nLast)) {

                bifR.push(r);

                bifX.push(x);

            }

        }

    }


    // 3. Setup Layout Profiles

    const commonLayout = {

        margin: { t: 40, b: 40, l: 50, r: 20 },

        hovermode: 'closest'

    };


    // 4. Main Update Loop Engine

    function updatePlots() {

        const rTarget = parseFloat(document.getElementById('r-slider').value);

        const x0 = parseFloat(document.getElementById('x-slider').value);

        const nCobweb = parseInt(document.getElementById('iter-slider').value);


        // Sync text labels

        document.getElementById('r-val').innerText = rTarget.toFixed(2);

        document.getElementById('x-val').innerText = x0.toFixed(2);

        document.getElementById('iter-val').innerText = nCobweb;


        // --- Panel 1: Render Bifurcation Plot ---

        const bifData = [

            { x: bifR, y: bifX, mode: 'markers', marker: { color: '#212529', size: 1, opacity: 0.15 }, type: 'scattergl', name: 'Attractor' },

            { x: [rTarget, rTarget], y:, mode: 'lines', line: { color: '#dc3545', width: 2, dash: 'dash' }, name: `r = ${rTarget}` }

        ];

        

        const layoutBif = {

            ...commonLayout,

            title: 'Bifurcation Diagram',

            xaxis: { title: 'Growth Parameter (r)', range: [2.4, 4.0] },

            yaxis: { title: 'Steady States (x)', range: [0, 1] },

            showlegend: false

        };

        Plotly.newPlot('bifurcation-plot', bifData, layoutBif);


        // --- Panel 2: Render Cobweb Plot ---

        // Construct curve grid mapping arrays

        const gridX = [], gridY = [];

        for (let i = 0; i <= 100; i++) {

            let px = i / 100;

            gridX.push(px);

            gridY.push(f(px, rTarget));

        }


        // Trace trajectory shifts

        const cobX = [], cobY = [];

        let currX = x0;

        for (let i = 0; i < nCobweb; i++) {

            let nextX = f(currX, rTarget);

            // Vertical movement

            cobX.push(currX, currX);

            cobY.push(currX, nextX);

            // Horizontal movement

            cobX.push(currX, nextX);

            cobY.push(nextX, nextX);

            currX = nextX;

        }


        const cobwebData = [

            { x: gridX, y: gridY, mode: 'lines', line: { color: '#0d6efd', width: 2.5 }, name: 'y=f(x)' },

            { x:, y:, mode: 'lines', line: { color: '#6c757d', width: 1.5, dash: 'dash' }, name: 'y=x' },

            { x: cobX, y: cobY, mode: 'lines', line: { color: '#dc3545', width: 1.2 }, name: 'Path' },

            { x: [x0], y: [x0], mode: 'markers', marker: { color: '#198754', size: 8 }, name: 'Start' }

        ];


        const layoutCob = {

            ...commonLayout,

            title: `Cobweb Trajectory Space`,

            xaxis: { title: 'x_n', range: [0, 1] },

            yaxis: { title: 'x_n+1', range: [0, 1] },

            yaxis: { scaleanchor: "x", scaleratio: 1 }, // Retain rigid geometric proportions

            showlegend: false

        };

        Plotly.newPlot('cobweb-plot', cobwebData, layoutCob);

    }


    // Attach real-time structural event listener updates to elements

    document.getElementById('r-slider').addEventListener('input', updatePlots);

    document.getElementById('x-slider').addEventListener('input', updatePlots);

    document.getElementById('iter-slider').addEventListener('input', updatePlots);


    // Initial Trigger Run

    updatePlots();

</script>
